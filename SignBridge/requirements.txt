# SignBridge dependencies.
#
# Added incrementally, one development milestone at a time.

# Milestone 1: webcam access + hand tracking
#
# mediapipe is pinned to the last release that still ships the stable legacy
# `mp.solutions.hands` API. Newer releases (0.10.35 and 1.x) removed it, and
# their replacement Tasks API aborts the process on Apple Silicon inside Metal.
mediapipe==0.10.21

# mediapipe 0.10.21 requires numpy<2, and opencv-python 5.x requires numpy>=2,
# so OpenCV is held at 4.x. mediapipe itself depends on opencv-contrib-python,
# so we pin that package rather than opencv-python to avoid installing two
# competing copies of cv2.
opencv-contrib-python>=4.8,<5

# Milestone 2: ASL sign classifier (random forest over landmark features).
# joblib ships with scikit-learn and is its documented way to persist models;
# it is listed because train.py and recognizer.py import it directly.
scikit-learn>=1.3,<2
joblib>=1.3

# Milestone 3: Gemini multimodal ASL recognition over a frame sequence.
# google-genai is the current unified Google AI SDK (`from google import genai`);
# the older google-generativeai package is superseded. It installs cleanly
# against the numpy<2 pin above.
google-genai>=1.0,<3
