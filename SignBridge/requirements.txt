# SignBridge dependencies.
#
# mediapipe is pinned to the last release that still ships the stable legacy
# `mp.solutions.hands` API. Newer releases (0.10.35 and 1.x) removed it, and
# their replacement Tasks API aborts the process on Apple Silicon inside Metal.
mediapipe==0.10.21

# mediapipe 0.10.21 requires numpy<2, and opencv-python 5.x requires numpy>=2,
# so OpenCV is held at 4.x. mediapipe itself depends on opencv-contrib-python,
# so we pin that package rather than opencv-python to avoid installing two
# competing copies of cv2.
opencv-contrib-python>=4.8,<5

# Optional: the unused Milestone 2 frame classifier still imports these.
scikit-learn>=1.3,<2
joblib>=1.3

# Live temporal classifier (local BiGRU). Pins stay below 2.6 so the wheel
# still installs against mediapipe's numpy<2 requirement.
torch>=2.1,<2.6

# ElevenLabs TTS. The live demo still talks HTTP via urllib; the audio
# diagnostic (`python -m src.speech.test_elevenlabs_audio`) uses the official
# SDK. convert() in 2.x returns Iterator[bytes].
elevenlabs>=2.50,<3

# ElevenLabs TTS playback. Audio is fetched with urllib (stdlib); this plays
# the PCM/WAV stream through PortAudio on macOS and Windows.
sounddevice>=0.4,<0.6

# Web front end. The browser owns the webcam and posts frames to a small Flask
# server that reuses the tracker and the temporal sequence model.
flask>=3,<4
