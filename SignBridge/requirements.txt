# Milestone 1: webcam access + hand tracking

# mediapipe is pinned to the last release that still ships the stable legacy
# `mp.solutions.hands` API. Newer releases (0.10.35 and 1.x) removed it, and
# their replacement Tasks API aborts the process on Apple Silicon inside Metal.
mediapipe==0.10.21

# mediapipe 0.10.21 requires numpy<2, and opencv-python 5.x requires numpy>=2,
# so OpenCV is held at 4.x. mediapipe itself depends on opencv-contrib-python,
# so we pin that package rather than opencv-python to avoid installing two
# competing copies of cv2.
opencv-contrib-python>=4.8,<5
